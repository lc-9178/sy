local function safeCall(fn, ...)
    local ok, a, b, c = pcall(fn, ...)
    if not ok then
        warn("[POLY HUB reconstruction] " .. tostring(a))
        return nil
    end
    return a, b, c
end

local function getCharacterParts()
    local character = LocalPlayer.Character
    if not character then return nil, nil, nil end
    local humanoid = character:FindFirstChildOfClass("Humanoid")
    local root = character:FindFirstChild("HumanoidRootPart")
    return character, humanoid, root
end

local function getRemote(name)
    return ReplicatedStorage:FindFirstChild(name)
end

local function fireRemote(name, ...)
    local remote = getRemote(name)
    if remote and remote.FireServer then
        remote:FireServer(...)
        return true
    end
    return false
end

local function normalizeSelection(value)
    if type(value) ~= "table" then return { value } end
    local result = {}
    for key, selected in pairs(value) do
        if type(key) == "number" then
            result[#result + 1] = selected
        elseif selected then
            result[#result + 1] = key
        end
    end
    return result
end

local function contains(list, wanted)
    for _, value in ipairs(normalizeSelection(list)) do
        if value == wanted then return true end
    end
    return false
end

local orientationAttachment
local orientationAlign

local function clearOrientation()
    if orientationAttachment then orientationAttachment:Destroy() end
    if orientationAlign then orientationAlign:Destroy() end
    orientationAttachment = nil
    orientationAlign = nil
end

local function ensureOrientation()
    local _, humanoid, root = getCharacterParts()
    if not humanoid or not root then return nil end
    clearOrientation()
    humanoid.AutoRotate = false
    orientationAttachment = Instance.new("Attachment", root)
    orientationAlign = Instance.new("AlignOrientation")
    orientationAlign.Mode = Enum.OrientationAlignmentMode.OneAttachment
    orientationAlign.Attachment0 = orientationAttachment
    orientationAlign.MaxTorque = math.huge
    orientationAlign.Responsiveness = 200
    orientationAlign.RigidityEnabled = false
    orientationAlign.Enabled = false
    orientationAlign.Parent = root
    return orientationAlign
end

local flyVelocity
local flyGyro

local function stopFly()
    if flyVelocity then flyVelocity:Destroy() end
    if flyGyro then flyGyro:Destroy() end
    flyVelocity = nil
    flyGyro = nil
    local _, humanoid = getCharacterParts()
    if humanoid then humanoid.PlatformStand = false end
end

local function startFly()
    local _, humanoid, root = getCharacterParts()
    if not humanoid or not root then return end
    stopFly()
    humanoid.PlatformStand = true
    flyVelocity = Instance.new("BodyVelocity")
    flyVelocity.MaxForce = Vector3.new(math.huge, math.huge, math.huge)
    flyVelocity.Velocity = Vector3.new(0, 0, 0)
    flyVelocity.Parent = root
    flyGyro = Instance.new("BodyGyro")
    flyGyro.MaxTorque = Vector3.new(math.huge, math.huge, math.huge)
    flyGyro.D = 100
    flyGyro.P = 10000
    flyGyro.CFrame = Workspace.CurrentCamera.CFrame
    flyGyro.Parent = root
end

local function findNearestLivingModel()
    -- [TRACE-INFERRED] The VM scans Workspace.Living, rejects player characters,
    -- and measures from HumanoidRootPart. Exact game-specific filters require live instances.
    local _, _, root = getCharacterParts()
    local living = Workspace:FindFirstChild("Living")
    if not root or not living then return nil end
    local nearest, nearestDistance
    for _, model in ipairs(living:GetChildren()) do
        if model:IsA("Model") and not Players:GetPlayerFromCharacter(model) then
            local targetRoot = model:FindFirstChild("HumanoidRootPart") or model.PrimaryPart
            local humanoid = model:FindFirstChildOfClass("Humanoid")
            if targetRoot and (not humanoid or humanoid.Health > 0) then
                local distance = (targetRoot.Position - root.Position).Magnitude
                if not nearestDistance or distance < nearestDistance then
                    nearest = model
                    nearestDistance = distance
                end
            end
        end
    end
    return nearest, nearestDistance
end

local function moveTowardTarget(target)
    -- [TRACE-INFERRED] Observable target scan/orientation preserved. The exact original
    -- offset was not reachable with dummy game objects, so this uses a conservative 4-stud offset.
    local _, _, root = getCharacterParts()
    local targetRoot = target and (target:FindFirstChild("HumanoidRootPart") or target.PrimaryPart)
    if not root or not targetRoot then return end
    local destination = targetRoot.CFrame * CFrame.new(0, 0, 4)
    if State.autoFarmMode == "Teleport" then
        root.CFrame = destination
    else
        local distance = (root.Position - destination.Position).Magnitude
        local speed = math.max(1, tonumber(State.farmMovementSpeed) or 100)
        TweenService:Create(root, TweenInfo.new(distance / speed, Enum.EasingStyle.Linear), { CFrame = destination }):Play()
    end
end

local function refreshConfigList()
    if not ensureConfigFolders() or type(listfiles) ~= "function" then return {} end
    local names = {}
    for _, path in ipairs(listfiles(CONFIG_DIR)) do
        local name = path:match("([^/\\]+)%.json$")
        if name then names[#names + 1] = name end
    end
    table.sort(names)
    local dropdown = Controls["Saved Configs"]
    if dropdown and dropdown.Refresh then dropdown:Refresh(names) end
    return names
end

RunService.Heartbeat:Connect(function()
    local _, humanoid, root = getCharacterParts()
    if humanoid then
        if State.walkSpeedEnabled then humanoid.WalkSpeed = State.walkSpeedValue end
        if State.jumpPowerEnabled then humanoid.UseJumpPower = true; humanoid.JumpPower = State.jumpPowerValue end
    end
    if State.fovEnabled then Workspace.CurrentCamera.FieldOfView = State.fovValue end
    if State.maxZoomEnabled then LocalPlayer.CameraMaxZoomDistance = State.maxZoomValue end

    if State.flyEnabled and root and flyVelocity and flyGyro then
        local camera = Workspace.CurrentCamera
        local direction = Vector3.new(0, 0, 0)
        if UserInputService:IsKeyDown(Enum.KeyCode.W) then direction = direction + camera.CFrame.LookVector end
        if UserInputService:IsKeyDown(Enum.KeyCode.S) then direction = direction - camera.CFrame.LookVector end
        if UserInputService:IsKeyDown(Enum.KeyCode.A) then direction = direction - camera.CFrame.RightVector end
        if UserInputService:IsKeyDown(Enum.KeyCode.D) then direction = direction + camera.CFrame.RightVector end
        if UserInputService:IsKeyDown(Enum.KeyCode.Space) then direction = direction + Vector3.new(0, 1, 0) end
        if UserInputService:IsKeyDown(Enum.KeyCode.LeftControl) then direction = direction - Vector3.new(0, 1, 0) end
        flyVelocity.Velocity = direction.Magnitude > 0 and direction.Unit * State.flySpeed or Vector3.new(0, 0, 0)
        flyGyro.CFrame = camera.CFrame
    end

    if State.astroFarmEnabled and root then
        local map = Workspace:FindFirstChild("Map")
        local circle = map and map:FindFirstChild("Circle")
        if circle then
            root.Velocity = Vector3.new(0, 0, 0)
            root.CFrame = CFrame.new(circle.Position + Vector3.new(0, 3, 0))
        end
    end
end)

local function loop(interval, callback)
    task.spawn(function()
        while task.wait(interval) do
            local ok, err = pcall(callback)
            if not ok then warn("[POLY HUB loop] " .. tostring(err)) end
        end
    end)
end

loop(0.10, function()
    if State.godModeEnabled then
        local _, humanoid = getCharacterParts()
        if humanoid and humanoid.MaxHealth > 0 and humanoid.Health / humanoid.MaxHealth * 100 <= State.godModeThreshold then
            humanoid.Health = 0
        end
    end
end)

loop(0.10, function()
    if not State.autoFarmEnabled and not State.faceNearestEnabled and not State.hitboxEnabled and not State.flushAuraEnabled then return end
    local target = findNearestLivingModel()
    updateStat("Current Target", target and target.Name or "Scanning...")
    if not target then return end
    local _, _, root = getCharacterParts()
    local targetRoot = target:FindFirstChild("HumanoidRootPart") or target.PrimaryPart
    if State.autoFarmEnabled then moveTowardTarget(target) end
    if State.faceNearestEnabled and orientationAlign and root and targetRoot then
        orientationAlign.Enabled = true
        orientationAlign.CFrame = CFrame.lookAt(root.Position, targetRoot.Position)
    end
    if State.hitboxEnabled and targetRoot then
        targetRoot.Size = Vector3.new(2, 2, 1) * State.hitboxMultiplier
        targetRoot.CanCollide = false
        targetRoot.Transparency = 0.7
    end
    if State.flushAuraEnabled and type(fireproximityprompt) == "function" then
        for _, object in ipairs(target:GetDescendants()) do
            if object:IsA("ProximityPrompt") then fireproximityprompt(object) end
        end
    end
end)
